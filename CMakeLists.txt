cmake_minimum_required(VERSION 3.20)
project(SpectacleOCR LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SOCR_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SOCR_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(PkgConfig REQUIRED)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Quick Qml)
find_package(KF6ConfigWidgets REQUIRED)
pkg_check_modules(TESSERACT REQUIRED IMPORTED_TARGET tesseract=5.5.3)
add_library(spectacle_ocr_hook SHARED native/hook.cpp native/settings.cpp native/formula_ui.cpp native/ui.qrc)
target_link_libraries(spectacle_ocr_hook PRIVATE PkgConfig::TESSERACT ${CMAKE_DL_LIBS} Qt6::Widgets Qt6::Quick Qt6::Qml KF6::ConfigWidgets)
target_compile_options(spectacle_ocr_hook PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ocr-host tests/host.cpp)
target_link_libraries(ocr-host PRIVATE PkgConfig::TESSERACT)
target_compile_options(ocr-host PRIVATE -Wall -Wextra -Wpedantic)
add_executable(settings-host tests/settings-host.cpp)
target_link_libraries(settings-host PRIVATE Qt6::Widgets KF6::ConfigWidgets)
add_library(ui-probe SHARED tests/ui-probe.cpp)
target_link_libraries(ui-probe PRIVATE Qt6::Widgets Qt6::Quick Qt6::Qml)
add_executable(download-host tests/download-host.cpp)
target_include_directories(download-host PRIVATE native)
target_link_libraries(download-host PRIVATE spectacle_ocr_hook Qt6::Widgets)
